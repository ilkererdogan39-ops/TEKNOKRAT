    <script>
        const DB_KEYS = { USERS: 'lms_users', ASSIGNMENTS: 'lms_assignments', LOGGED_IN_USER: 'lms_session' };

        function getData(key) { return JSON.parse(localStorage.getItem(key)) || []; }
        function saveData(key, data) { localStorage.setItem(key, JSON.stringify(data)); }

        function showToast(msg, type='success') {
            const t = document.getElementById('toast'); t.textContent=msg; t.style.background=type==='error'?'#e74c3c':'#27ae60'; t.classList.add('show'); setTimeout(()=>t.classList.remove('show'),3000);
        }

        function hideAllSections() {
            ['landing-page','admin-login','admin-dashboard','participant-login','participant-dashboard','video-modal','add-user-modal'].forEach(id=>document.getElementById(id).classList.add('hidden'));
        }

        function goHome() { hideAllSections(); document.getElementById('landing-page').classList.remove('hidden'); logout(true); }
        function logout(soft=false) { localStorage.removeItem(DB_KEYS.LOGGED_IN_USER); if(!soft) { goHome(); showToast('Çıkış yapıldı.'); } }
        function showLogin(role) {
            hideAllSections();
            if(role==='admin') document.getElementById('admin-login').classList.remove('hidden');
            else { document.getElementById('participant-login').classList.remove('hidden'); resetLoginFlow(); }
        }

        // --- ADMIN LOGIC ---
        function adminLogin() {
            const u = document.getElementById('admin-user').value;
            const p = document.getElementById('admin-pass').value;
            if (u === 'admin' && p === 'egitim123') {
                hideAllSections(); document.getElementById('admin-dashboard').classList.remove('hidden'); loadAdminData();
            } else showToast('Hatalı bilgiler!', 'error');
        }

        function switchAdminTab(t) {
            ['users','trainings','reports'].forEach(x=>{document.getElementById('tab-'+x).classList.add('hidden'); document.getElementById('btn-tab-'+x).classList.remove('active');});
            document.getElementById('tab-'+t).classList.remove('hidden'); document.getElementById('btn-tab-'+t).classList.add('active');
            if(t==='users') renderUserTable();
            if(t==='trainings') renderAssignments();
            if(t==='reports') loadAdminReports();
        }

        function loadAdminData() { renderUserTable(); renderAssignments(); populateTrainingUserSelect(); }

        // --- YÖNTEM: OLAY DELEGASYONU (EVENT DELEGATION) ---
        // Tüm tablo butonlarını tek bir yerden dinliyoruz. Bu en sağlam yöntemdir.
        
        document.getElementById('user-table-body').addEventListener('click', function(e) {
            // Tıklanan bir buton mu?
            if (e.target.tagName === 'BUTTON') {
                const btn = e.target;
                const sicil = btn.getAttribute('data-sicil');

                // Silme butonu mu?
                if (btn.classList.contains('btn-delete')) {
                    if(confirm("Bu kaydı silmek istiyor musunuz?")) {
                        let users = getData(DB_KEYS.USERS);
                        users = users.filter(u => u.sicil !== sicil);
                        saveData(DB_KEYS.USERS, users);
                        
                        // Atamaları da temizle
                        let assign = getData(DB_KEYS.ASSIGNMENTS);
                        assign = assign.filter(a => a.userSicil !== sicil);
                        saveData(DB_KEYS.ASSIGNMENTS, assign);
                        
                        renderUserTable();
                        populateTrainingUserSelect();
                        showToast('Kayıt silindi.');
                    }
                }
                
                // Şifre sıfırlama butonu mu?
                if (btn.classList.contains('btn-reset')) {
                    let users = getData(DB_KEYS.USERS);
                    const user = users.find(u => u.sicil === sicil);
                    if(user) {
                        user.password = null;
                        saveData(DB_KEYS.USERS, users);
                        renderUserTable();
                        showToast('Şifre sıfırlandı.');
                    }
                }
            }
        });

        // Eğitim Silme Delegasyonu
        document.getElementById('assignments-table-body').addEventListener('click', function(e) {
            if (e.target.tagName === 'BUTTON') {
                const id = e.target.getAttribute('data-id');
                if (confirm("Bu eğitimi silmek istiyor musunuz?")) {
                    let assign = getData(DB_KEYS.ASSIGNMENTS);
                    assign = assign.filter(a => a.id !== id);
                    saveData(DB_KEYS.ASSIGNMENTS, assign);
                    renderAssignments();
                    showToast('Eğitim silindi.');
                }
            }
        });

        function renderUserTable() {
            const users = getData(DB_KEYS.USERS);
            const tbody = document.getElementById('user-table-body');
            tbody.innerHTML = '';
            users.forEach(u => {
                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td>${u.sicil}</td>
                    <td>${u.name}</td>
                    <td>${u.dept}</td>
                    <td>${u.email}</td>
                    <td>${u.password ? '<span style="color:green">●●●●</span>' : '<span style="color:red">Belirlenmedi</span>'}</td>
                    <td>
                        <button class="danger btn-reset" data-sicil="${u.sicil}">Şifre Sıfırla</button>
                        <button class="danger btn-delete" data-sicil="${u.sicil}">Sil</button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

        function toggleAddUserModal(show) {
            const m = document.getElementById('add-user-modal');
            if(show) {
                m.classList.remove('hidden');
                ['new-user-sicil','new-user-name','new-user-dept','new-user-email'].forEach(id=>document.getElementById(id).value='');
            } else m.classList.add('hidden');
        }

        function saveNewUser() {
            const s=document.getElementById('new-user-sicil').value.trim();
            const n=document.getElementById('new-user-name').value.trim();
            const d=document.getElementById('new-user-dept').value.trim();
            const e=document.getElementById('new-user-email').value.trim();
            if(!s||!n||!e) return showToast('Alanları doldurun.','error');
            let users=getData(DB_KEYS.USERS);
            if(users.find(u=>u.sicil===s)) return showToast('Kayıt var.','error');
            users.push({sicil:s,name:n,dept:d,email:e,password:null});
            saveData(DB_KEYS.USERS,users);
            renderUserTable(); populateTrainingUserSelect(); toggleAddUserModal(false); showToast('Eklendi.');
        }

        function importCSV(input) {
            const file = input.files[0]; if(!file) return;
            const reader = new FileReader();
            reader.onload = function(e) {
                const lines = e.target.result.split('\n');
                let users = getData(DB_KEYS.USERS);
                let c=0;
                for(let i=1; i<lines.length; i++) {
                    const cols = lines[i].split(',');
                    if(cols.length>=4) {
                        const s=cols[0].trim();
                        if(!users.find(u=>u.sicil===s)) {
                            users.push({sicil:s,name:cols[1].trim(),dept:cols[2].trim(),email:cols[3].trim(),password:null});
                            c++;
                        }
                    }
                }
                saveData(DB_KEYS.USERS,users); renderUserTable(); populateTrainingUserSelect(); showToast(c+' kişi aktarıldı.');
            };

        function getEmbedUrl(url) {
            if(url.includes('youtube.com') || url.includes('youtu.be')) {
                let v='';
                if(url.includes('v=')) v=url.split('v=')[1].split('&')[0];
                else if(url.includes('youtu.be/')) v=url.split('youtu.be/')[1];
                return `https://www.youtube.com/embed/${v}`;
            } else if(url.includes('vimeo.com')) return `https://player.vimeo.com/video/${url.split('.com/')[1]}`;
            return '';
        }

        function assignTraining() {
            const t=document.getElementById('training-title').value;
            const url=document.getElementById('training-url').value;
            const sel=document.getElementById('training-target-users');
            const su=Array.from(sel.selectedOptions).map(o=>o.value);
            if(!t||!url||su.length===0) return showToast('Eksik bilgi.','error');
            const eu=getEmbedUrl(url); if(!eu) return showToast('Geçersiz link.','error');
            let assign=getData(DB_KEYS.ASSIGNMENTS);
            let users=getData(DB_KEYS.USERS);
            su.forEach(sic=>{
                const u=users.find(x=>x.sicil===sic);
                assign.push({id:Date.now()+Math.random().toString(), userSicil:sic, userName:u.name, title:t, url:url, embedUrl:eu, progress:0, completed:false, assignedDate:new Date().toLocaleDateString('tr-TR')});
            });
            saveData(DB_KEYS.ASSIGNMENTS, assign);
            document.getElementById('training-title').value=''; document.getElementById('training-url').value=''; sel.selectedIndex=-1;
            renderAssignments(); showToast('Atandı.');
        }

        function loadUserReport(sicil) {
            const ar=document.getElementById('report-card-area');
            if(!sicil) return ar.innerHTML='<p>Kişi seçin.</p>';
            const u=getData(DB_KEYS.USERS).find(x=>x.sicil===sicil);
            const a=getData(DB_KEYS.ASSIGNMENTS).filter(x=>x.userSicil===sicil);
            let h=`<h3>${u.name} - Karnesi</h3><p>${u.dept} | ${u.email}</p>`;
            if(a.length) {
                h+='<table><thead><tr><th>Eğitim</th><th>Tarih</th><th>Durum</th></tr></thead><tbody>';

        }

        function verifyIdentity() {
            const s=document.getElementById('part-login-sicil').value;
            const e=document.getElementById('part-login-email').value;
            const u=getData(DB_KEYS.USERS).find(x=>x.sicil===s && x.email===e);

            else { document.getElementById('step-verify').classList.add('hidden'); document.getElementById('step-set-password').classList.remove('hidden'); }

        function setPassword() {
            const p=document.getElementById('part-new-pass').value;
            if(p.length<4) return showToast('Şifre kısa.','error');
            let us=getData(DB_KEYS.USERS); const idx=us.findIndex(x=>x.sicil===tempSicil);
            us[idx].password=p; saveData(DB_KEYS.USERS,us); showToast('Ayarlandı.'); loginUser(us[idx]);
        }
        function participantLogin() {
            const p=document.getElementById('part-input-pass').value;
            const u=getData(DB_KEYS.USERS).find(x=>x.sicil===tempSicil && x.password===p);
            if(u) loginUser(u); else showToast('Hatalı şifre.','error');
        }

        }

        let currentAssignmentId=null;

        function openVideo(id) {
            const a=getData(DB_KEYS.ASSIGNMENTS).find(x=>x.id===id); if(!a) return;
            currentAssignmentId=id;
            document.getElementById('modal-title').textContent=a.title;
            document.getElementById('video-frame').src=a.embedUrl+"?autoplay=1";
            document.getElementById('video-progress-bar').style.width=a.progress+'%';
            document.getElementById('progress-text').textContent=a.progress+'%';
            document.getElementById('complete-training-btn').disabled=a.progress<80;
            document.getElementById('video-modal').classList.remove('hidden');
        }
        function closeVideoModal() { document.getElementById('video-modal').classList.add('hidden'); document.getElementById('video-frame').src=""; }
        function simulateProgress(amount) {
            let a=getData(DB_KEYS.ASSIGNMENTS);
            const idx=a.findIndex(x=>x.id===currentAssignmentId);
            if(idx>-1) {
                let p=a[idx].progress+amount;
                if(p>100)p=100;
                a[idx].progress=p;
                saveData(DB_KEYS.ASSIGNMENTS,a);
                document.getElementById('video-progress-bar').style.width=p+'%';
                document.getElementById('progress-text').textContent=p+'%';
                if(p>=80) { document.getElementById('complete-training-btn').disabled=false; showToast('%80 tamamlandı.'); }
            }
        }
        function completeTraining() {
            let a=getData(DB_KEYS.ASSIGNMENTS);
            const idx=a.findIndex(x=>x.id===currentAssignmentId);
            if(idx>-1) {
                a[idx].progress=100; a[idx].completed=true;
                saveData(DB_KEYS.ASSIGNMENTS,a);
                showToast('Tebrikler!'); closeVideoModal();
                const u=JSON.parse(localStorage.getItem(DB_KEYS.LOGGED_IN_USER));
                renderParticipantTrainings(u.sicil);
            }
        }
